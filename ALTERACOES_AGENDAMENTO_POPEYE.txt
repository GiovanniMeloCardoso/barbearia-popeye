Data: 17/10/2026
Objetivo da alteração:
Resolver o problema no celular em que o agendamento não era salvo no banco de dados
porque o aplicativo do WhatsApp abria imediatamente e minimizava o navegador, 
interrompendo a requisição.
Agora o fluxo foi dividido em 2 etapas:
1) O cliente clica em "Confirmar Agendamento", salvando na hora no banco via AJAX.
2) Em seguida, abre a tela de sucesso com a instrução:
   "Envie uma mensagem ao barbeiro para informá-lo sobre o agendamento",
   com o botão para abrir o WhatsApp sem risco de perder o agendamento.

def index(request):
    servicos = Servico.objects.all()
    
    if request.method == 'POST':
        nome = request.POST.get('nome')
        telefone = request.POST.get('telefone')
        servicos_ids = request.POST.getlist('servicos')
        data = request.POST.get('data')
        hora = request.POST.get('hora')
        token = request.POST.get('token')
        
        try:
            data_obj = datetime.strptime(data, '%Y-%m-%d').date()
            hoje = date.today()
            
            if data_obj < hoje or data_obj.weekday() == 6:
                messages.error(request, 'Data inválida ou domingo.')
                return redirect('index')

            se_ocupado = Agendamento.objects.filter(data=data, hora=hora).exists()
            
            if se_ocupado:
                messages.error(request, 'Este horário acabou de ser reservado. Escolha outro.')
            else:
                agendamento = Agendamento.objects.create(
                    nome_cliente=nome,
                    telefone=telefone,
                    data=data,
                    hora=hora,
                    token=token
                )
                for s_id in servicos_ids:
                    agendamento.servicos.add(s_id)
                    
                messages.success(request, 'Agendamento realizado com sucesso! Te esperamos.')
            return redirect('index')
        except Exception as e:
            messages.error(request, 'Ocorreu um erro ao agendar. Tente novamente.')

    return render(request, 'agendamento/index.html', {'servicos': servicos})

def index(request):
    servicos = Servico.objects.all()
    
    if request.method == 'POST':
        nome = request.POST.get('nome')
        telefone = request.POST.get('telefone')
        servicos_ids = request.POST.getlist('servicos')
        data = request.POST.get('data')
        hora = request.POST.get('hora')
        token = request.POST.get('token')
        
        # Identifica se é uma requisição assíncrona (AJAX/Fetch)
        is_ajax = request.headers.get('x-requested-with') == 'XMLHttpRequest' or 'application/json' in request.headers.get('accept', '')
        
        try:
            data_obj = datetime.strptime(data, '%Y-%m-%d').date()
            hoje = date.today()
            
            if data_obj < hoje or data_obj.weekday() == 6:
                if is_ajax:
                    return JsonResponse({'sucesso': False, 'mensagem': 'Data inválida ou domingo.'}, status=400)
                messages.error(request, 'Data inválida ou domingo.')
                return redirect('index')

            se_ocupado = Agendamento.objects.filter(data=data, hora=hora).exists()
            
            if se_ocupado:
                if is_ajax:
                    return JsonResponse({'sucesso': False, 'mensagem': 'Este horário acabou de ser reservado. Escolha outro.'}, status=400)
                messages.error(request, 'Este horário acabou de ser reservado. Escolha outro.')
            else:
                agendamento = Agendamento.objects.create(
                    nome_cliente=nome,
                    telefone=telefone,
                    data=data,
                    hora=hora,
                    token=token
                )
                for s_id in servicos_ids:
                    agendamento.servicos.add(s_id)
                    
                if is_ajax:
                    return JsonResponse({'sucesso': True, 'mensagem': 'Agendamento realizado com sucesso! Te esperamos.'})
                messages.success(request, 'Agendamento realizado com sucesso! Te esperamos.')
            return redirect('index')
        except Exception as e:
            if is_ajax:
                return JsonResponse({'sucesso': False, 'mensagem': 'Ocorreu um erro ao agendar. Tente novamente.'}, status=500)
            messages.error(request, 'Ocorreu um erro ao agendar. Tente novamente.')

    return render(request, 'agendamento/index.html', {'servicos': servicos})
